cmake_minimum_required(VERSION 3.20)
project(pxa_video_decoder_test CXX)
set(CMAKE_CXX_STANDARD 20)
set(root "${CMAKE_CURRENT_LIST_DIR}/..")
set(vendor "${root}/vendor/openh264")
file(GLOB sources "${vendor}/codec/common/src/*.cpp"
                  "${vendor}/codec/decoder/core/src/*.cpp"
                  "${vendor}/codec/decoder/plus/src/*.cpp")
add_library(openh264_decoder STATIC ${sources})
target_include_directories(openh264_decoder PUBLIC "${vendor}/codec/api" "${vendor}/codec/api/wels"
    "${vendor}/codec/common/inc" "${vendor}/codec/decoder/core/inc" "${vendor}/codec/decoder/plus/inc")
target_compile_options(openh264_decoder PRIVATE -O3 -fno-strict-aliasing)
find_package(Threads REQUIRED)
target_link_libraries(openh264_decoder PUBLIC Threads::Threads)
add_executable(decoder_test decoder_test.cpp)
target_link_libraries(decoder_test PRIVATE openh264_decoder)
target_include_directories(decoder_test PRIVATE "${root}/src" "${root}/../../../deps/pxa-system/libpxa/include")
target_compile_options(decoder_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
enable_testing()
add_executable(limits_test limits_test.cpp)
target_include_directories(limits_test PRIVATE "${root}/src" "${root}/../../../deps/pxa-system/libpxa/include")
target_compile_options(limits_test PRIVATE -UNDEBUG -Wall -Wextra -Werror -fsanitize=address,undefined)
target_link_options(limits_test PRIVATE -fsanitize=address,undefined)
add_test(NAME h264_sps_resource_limits COMMAND limits_test)
add_test(NAME high_profile_decoder COMMAND decoder_test "${CMAKE_CURRENT_LIST_DIR}/fixtures/high-640x360.h264")
set(PXA_VIDEO_CONTROL_OVERLAY "" CACHE FILEPATH "Generated ESP control source for allocation-failure regression")
if(PXA_VIDEO_CONTROL_OVERLAY)
    add_executable(control_memory_test control_memory_test.cpp "${PXA_VIDEO_CONTROL_OVERLAY}")
    get_filename_component(overlay_dir "${PXA_VIDEO_CONTROL_OVERLAY}" DIRECTORY)
    list(FILTER sources EXCLUDE REGEX "/decoder.cpp$")
    list(FILTER sources EXCLUDE REGEX "/pic_queue.cpp$")
    set_property(TARGET openh264_decoder PROPERTY SOURCES ${sources} "${overlay_dir}/decoder.cpp" "${overlay_dir}/pic_queue.cpp")
    target_include_directories(openh264_decoder PRIVATE "${root}/src")
    target_sources(control_memory_test PRIVATE "${overlay_dir}/WelsThreadLib.cpp" "${overlay_dir}/memory_align.cpp")
    target_include_directories(control_memory_test PRIVATE "${root}/src" "${overlay_dir}")
    target_link_libraries(control_memory_test PRIVATE openh264_decoder)
    target_compile_options(control_memory_test PRIVATE -UNDEBUG -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(control_memory_test PRIVATE -fsanitize=address,undefined)
    add_test(NAME control_allocation_failure COMMAND control_memory_test "${CMAKE_CURRENT_LIST_DIR}/fixtures/high-640x360.h264")
endif()

add_executable(random_access_test random_access_test.cpp)
target_link_libraries(random_access_test PRIVATE openh264_decoder)
target_include_directories(random_access_test PRIVATE "${root}/src" "${root}/../../../deps/pxa-system/libpxa/include")
target_compile_options(random_access_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME idr_jump_without_reset COMMAND random_access_test "${CMAKE_CURRENT_LIST_DIR}/fixtures/high-640x360.h264")

add_executable(rgb565_test rgb565_test.cpp)
target_include_directories(rgb565_test PRIVATE "${root}/src")
target_compile_options(rgb565_test PRIVATE -O2 -UNDEBUG -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(rgb565_test PRIVATE -fsanitize=address,undefined)
add_test(NAME rgb565_exact_conversion COMMAND rgb565_test)

add_executable(decoder_thread_test thread_output_test.cpp)
target_link_libraries(decoder_thread_test PRIVATE openh264_decoder)
target_include_directories(decoder_thread_test PRIVATE "${root}/src" "${root}/../../../deps/pxa-system/libpxa/include")
target_compile_options(decoder_thread_test PRIVATE -O2 -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME high_profile_two_workers COMMAND decoder_thread_test "${CMAKE_CURRENT_LIST_DIR}/fixtures/high-640x360.h264")
set_tests_properties(high_profile_two_workers PROPERTIES TIMEOUT 30)
