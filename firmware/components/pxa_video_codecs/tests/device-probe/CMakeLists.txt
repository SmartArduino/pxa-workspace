cmake_minimum_required(VERSION 3.20)
project(pxa_video_probe C)
list(APPEND CMAKE_MODULE_PATH "${PXA_CMAKE_MODULE_DIR}")
include(PxaGuest)
set(fixture_header "${CMAKE_CURRENT_BINARY_DIR}/fixtures.h")
file(WRITE "${fixture_header}" "#include <stdint.h>\n")
foreach(pair "high;high-640x360.h264" "jpeg;colors-320x180.jpg" "aac;tone-44100-stereo.aac")
    list(GET pair 0 name)
    list(GET pair 1 filename)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/${filename}" hex HEX)
    string(REGEX REPLACE "(..)" "0x\\1," data "${hex}")
    file(APPEND "${fixture_header}" "static const uint8_t ${name}[] = {${data}};\n")
endforeach()
pxa_add_component(main COMPONENT_ID main SOURCES main.c INCLUDE_DIRS "${CMAKE_CURRENT_BINARY_DIR}")
