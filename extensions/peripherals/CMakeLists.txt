cmake_minimum_required(VERSION 3.20)
project(pxa_peripherals VERSION 0.3.0 LANGUAGES C CXX)
set(PXA_SYSTEM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../deps/pxa-system" CACHE PATH "PXA Host/Guest SDK checkout")
if(NOT TARGET pxa)
    add_subdirectory("${PXA_SYSTEM_ROOT}/libpxa" "${CMAKE_CURRENT_BINARY_DIR}/libpxa")
endif()
add_library(pxa_peripherals STATIC src/peripherals.c)
target_include_directories(pxa_peripherals PUBLIC include)
target_link_libraries(pxa_peripherals PUBLIC pxa)
target_compile_features(pxa_peripherals PRIVATE c_std_99)
target_compile_options(pxa_peripherals PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(sdk/Peripherals.cmake)
option(PXA_PERIPHERALS_TESTS "Build the extension integration tests" ON)
if(PXA_PERIPHERALS_TESTS)
    enable_testing()
    add_executable(peripherals_host_test tests/host_test.c)
    target_link_libraries(peripherals_host_test PRIVATE pxa_peripherals)
    target_compile_options(peripherals_host_test PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
    add_test(NAME peripherals_host COMMAND peripherals_host_test)
    add_executable(peripherals_async_host_test tests/async_host_test.c)
    target_link_libraries(peripherals_async_host_test PRIVATE pxa_peripherals)
    target_compile_options(peripherals_async_host_test PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
    add_test(NAME peripherals_async_host COMMAND peripherals_async_host_test)
    add_executable(peripherals_sdk_test tests/sdk_test.cpp
        "${PXA_SYSTEM_ROOT}/sdk/guest-cpp/src/runtime.cpp")
    target_link_libraries(peripherals_sdk_test PRIVATE pxa::peripherals)
    target_include_directories(peripherals_sdk_test PRIVATE
        "${PXA_SYSTEM_ROOT}/sdk/guest-cpp/include"
        "${PXA_SYSTEM_ROOT}/sdk/testing")
    target_compile_options(peripherals_sdk_test PRIVATE -std=c++2c
        -fno-exceptions -fno-rtti -Wall -Wextra -Werror -Wno-attributes -UNDEBUG)
    add_test(NAME peripherals_sdk COMMAND peripherals_sdk_test)
    add_test(NAME peripherals_sdk_v01_host COMMAND peripherals_sdk_test v01)
    add_executable(vision_sdk_test tests/vision_sdk_test.cpp)
    target_link_libraries(vision_sdk_test PRIVATE pxa::peripherals)
    target_include_directories(vision_sdk_test PRIVATE "${PXA_SYSTEM_ROOT}/sdk/guest-cpp/include")
    target_compile_options(vision_sdk_test PRIVATE -std=c++2c -fno-exceptions -fno-rtti
        -Wall -Wextra -Werror -Wno-attributes -UNDEBUG)
    add_test(NAME vision_sdk COMMAND vision_sdk_test)
endif()
