cmake_minimum_required(VERSION 3.20)
project(pxa_wifi_location LANGUAGES C)
set(PXA_SYSTEM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../deps/pxa-system" CACHE PATH "PXA System checkout")
set(PXA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PXA_BUILD_ADAPTERS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PXA_SYSTEM_ROOT}/libpxa" libpxa)
add_library(pxa_wifi_location src/wifi_location.c)
target_include_directories(pxa_wifi_location PUBLIC include)
target_link_libraries(pxa_wifi_location PUBLIC pxa)
target_compile_options(pxa_wifi_location PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_executable(wifi_location_test tests/host_test.c)
target_link_libraries(wifi_location_test PRIVATE pxa_wifi_location)
target_compile_options(wifi_location_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME wifi_location COMMAND wifi_location_test)
