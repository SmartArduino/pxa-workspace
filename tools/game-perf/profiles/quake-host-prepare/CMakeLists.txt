cmake_minimum_required(VERSION 3.20)
project(pxa_quake_host_prepare C CXX)
# Keep comparison defaults independent of the public SD package defaults.
foreach(feature PXA_QUAKE_LIGHTMAP_CACHE PXA_RECROF_BULK_READ_COPY PXA_RECROF_PACKED_GAME_CODEC)
    if(NOT DEFINED ${feature})
        set(${feature} OFF CACHE BOOL "Private comparison default")
    endif()
endforeach()
set(PXA_RECROF_MAIN_LOOP ON)
set(PXA_RECROF_NATIVE_PCM ON)
set(PXA_RECROF_HOST_BLIT ON CACHE BOOL "" FORCE)
set(PXA_QUAKE_HOST_AUDIO_MIX OFF CACHE BOOL "" FORCE)
set(PXA_QUAKE_RENDER_PROBE ON CACHE BOOL "" FORCE)
set(PXA_QUAKE_FRAME_SPANS ON CACHE BOOL "" FORCE)
set(PXA_QUAKE_FRAME_BULK_COPY ON CACHE BOOL "" FORCE)
set(PXA_QUAKE_LIT_CACHE ON CACHE BOOL "Exact Guest reference and fallback" FORCE)
include("${CMAKE_CURRENT_LIST_DIR}/../quake-source.cmake")
pxa_profile_add_quake()
